cmake_minimum_required(VERSION 3.20)
project(nicotine LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_path(FFI_INCLUDE_DIR ffi.h PATH_SUFFIXES ffi REQUIRED)
find_library(FFI_LIBRARY ffi REQUIRED)
set(Q vendor/quickjs)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/bootstrap.h
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed.py
            ${CMAKE_CURRENT_SOURCE_DIR}/js/bootstrap.js ${CMAKE_CURRENT_BINARY_DIR}/bootstrap.h
    DEPENDS js/bootstrap.js tools/embed.py)
add_executable(nicotine src/runtime.c ${CMAKE_CURRENT_BINARY_DIR}/bootstrap.h
    ${Q}/quickjs.c ${Q}/dtoa.c ${Q}/libregexp.c ${Q}/libunicode.c ${Q}/cutils.c)
target_include_directories(nicotine PRIVATE ${Q} ${FFI_INCLUDE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(nicotine PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_link_libraries(nicotine PRIVATE ${FFI_LIBRARY})
if(WIN32)
    # MinGW-w64 toolchain; QuickJS is not an MSVC target in this project.
    target_sources(nicotine PRIVATE src/platform_windows.c)
    target_link_libraries(nicotine PRIVATE winhttp)
    add_library(clock-win32 SHARED src/clock_win32.c)
    target_link_libraries(clock-win32 PRIVATE user32 gdi32)
    set_target_properties(clock-win32 PROPERTIES PREFIX "")
else()
    find_package(CURL REQUIRED)
    find_package(Threads REQUIRED)
    target_sources(nicotine PRIVATE src/platform_posix.c)
    target_link_libraries(nicotine PRIVATE CURL::libcurl Threads::Threads ${CMAKE_DL_LIBS} m)
    if(APPLE)
        enable_language(OBJC)
        add_library(clock-macos SHARED src/clock_macos.m)
        target_compile_options(clock-macos PRIVATE -fobjc-arc)
        target_link_libraries(clock-macos PRIVATE "-framework Cocoa")
        set_target_properties(clock-macos PROPERTIES PREFIX "")
    endif()
endif()
option(NICOTINE_GTK "Build the native GTK clock example" OFF)
option(NICOTINE_QT "Build the native Qt clock example" OFF)
if(NICOTINE_GTK)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
    add_library(clock-gtk SHARED src/clock_gtk.c)
    target_link_libraries(clock-gtk PRIVATE PkgConfig::GTK)
    set_target_properties(clock-gtk PROPERTIES PREFIX "")
endif()
if(NICOTINE_QT)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_library(clock-qt SHARED src/clock_qt.cpp)
    target_link_libraries(clock-qt PRIVATE Qt6::Widgets)
    set_target_properties(clock-qt PROPERTIES PREFIX "")
endif()
